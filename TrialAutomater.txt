    .addSeparator()
    .addItem('⏰ 5. Enable Automated Background Trigger (Hourly 24/7)', 'installHourlyTrigger')
    .addItem('🛑 6. Remove Hourly Triggers (Enable Spontaneous Mode)', 'removeHourlyTriggers')
    .addItem('🔍 7. Check Gmail Quota Remaining', 'checkQuota')
    .addToUi();

function installHourlyTrigger() {
  removeHourlyTriggers(); // Purge old triggers first to prevent duplicate trigger runs
  ScriptApp.newTrigger("processAllLifecycleEmails")
    .timeBased()
    .everyHours(1)
    .create();
  if (SpreadsheetApp.getActiveSpreadsheet()) {
    SpreadsheetApp.getActiveSpreadsheet().toast("Hourly background trigger installed! Expiry and lifecycle emails will now dispatch automatically 24/7.", "Automation Active", 6);
  }
}

function removeHourlyTriggers() {
  var triggers = ScriptApp.getProjectTriggers();
  var count = 0;
  for (var i = 0; i < triggers.length; i++) {
    var fn = triggers[i].getHandlerFunction();
    if (fn === "processAllLifecycleEmails" || fn === "processTrialEmails" || fn === "installHourlyTrigger") {
      ScriptApp.deleteTrigger(triggers[i]);
      count++;
    }
  }
  SpreadsheetApp.getActiveSpreadsheet().toast("Removed " + count + " hourly background trigger(s). Spontaneous real-time email dispatch is active.", "Spontaneous Mode Active", 6);
}

function checkQuota() {
  var remaining = MailApp.getRemainingDailyQuota();
  var user = Session.getEffectiveUser().getEmail();
  SpreadsheetApp.getUi().alert("Executing Account: " + user + "\nGmail/MailApp Remaining Daily Quota: " + remaining + " emails.");
}

/**
 * ─────────────────────────────────────────────────────────────────────────────
 * 3. REAL-TIME EVENT WEBHOOK (Instant Event Router + Anti-Spam Engine)
 * ─────────────────────────────────────────────────────────────────────────────
 */
function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
  } catch (err) {
    return jsonResponse({ status: "error", message: "Server busy" });
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = getRawDataSheet(ss);

    if (sheet.getLastRow() === 0) {
      createStandardHeaders(sheet);
      SpreadsheetApp.flush();
    }

    var rawBody = (e && e.postData && e.postData.contents) ? e.postData.contents : "";
    var payload = {};
    if (rawBody) {
      try {
        payload = JSON.parse(rawBody);
      } catch (_) {
        payload = (e && e.parameter) || {};
      }
    } else if (e && e.parameter) {
      payload = e.parameter;
    }

    var now = new Date();
    var timestampStr = formatToStandardIst(now);
    var email = String(payload.email || (e && e.parameter && e.parameter.email) || "").trim().toLowerCase();
    var product = payload.product || (e && e.parameter && e.parameter.product) || "ReturnPilot-GST Return Downloader";
    var term = payload.term || (e && e.parameter && e.parameter.term) || "";
    var rawAmount = String(payload.amount || (e && e.parameter && e.parameter.amount) || "").replace(/[^\d.]/g, "");
    var amount = rawAmount ? parseFloat(rawAmount) : "";
    var status = String(payload.status || (e && e.parameter && e.parameter.status) || "EVENT").toUpperCase();
    var source = payload.source || (e && e.parameter && e.parameter.source) || "Website";
    var device = payload.device || (e && e.parameter && e.parameter.device) || "Browser";
    var details = payload.details || (e && e.parameter && e.parameter.details) || "";

    // Error Telemetry Classification
    var isErrorLog = status === "ERROR_LOG" || status === "CRASH_REPORT" || source.toLowerCase().includes("error") || details.toLowerCase().includes("error_stack");
    if (isErrorLog) {
      var cleanDetails = String(details || payload.error || "Runtime Exception").replace(/[\r\n]+/g, " | ").slice(0, 350);
      sheet.appendRow([
        timestampStr,
        email || "anonymous_client",
        product,
        payload.returnType || payload.term || "Diagnostics",
        "",
        "ERROR_LOG",
        source || "Extension Error",
        device || (payload.v || payload.version ? "v" + (payload.v || payload.version) : "Browser"),
        cleanDetails,
        "LOGGED",
        "SKIPPED",
        "SKIPPED"
      ]);
      SpreadsheetApp.flush();
      return jsonResponse({ status: "success", message: "Error log recorded" });
    }

    // Event Classification
    var isFeedbackForm = status === STATUS.FEEDBACK_SUBMITTED || 
                         source.toLowerCase().includes("uninstall form") || 
                         term.toLowerCase().includes("uninstall feedback") ||
                         details.toLowerCase().includes("reason:") ||
                         details.toLowerCase().includes("rating:");

    var isUninstallHook = !isFeedbackForm && (
      status.includes("UNINSTALL") || 
      source.toLowerCase().includes("uninstall") || 
      details.toLowerCase().includes("extension removed") ||
      term.toLowerCase().includes("extension uninstalled")
    );

    // Normalize Status
    if (isFeedbackForm && status !== STATUS.FEEDBACK_SUBMITTED) {
      status = STATUS.FEEDBACK_SUBMITTED;
    } else if (isUninstallHook && !status.includes("UNINSTALL")) {
      status = STATUS.UNINSTALL_PAGE_OPENED;
    }

    // Anti-Piracy & Build Integrity Verification
    var rawVersion = String(payload.v || payload.version || (e && e.parameter && (e.parameter.v || e.parameter.version)) || "").trim();
    var rawTier = String(payload.tier || (e && e.parameter && e.parameter.tier) || "").trim().toUpperCase();
    var rawInstalledAt = String(payload.installed_at || (e && e.parameter && e.parameter.installed_at) || "").trim();
    var isPiratedBuild = false;
    var piracyNotes = [];

    const AUTHORIZED_VERSIONS = ["4.0.0", "4.0.1", "4.0.2", "4.0.3", "4.0.4", "4.1.0", "4.1.1", "4.1.2", "4.1.3", "4.1.4", "4.1.5", "4.1.6", "4.1.7", "4.1.8", "4.1.9", "4.2.0", "4.2.1", "4.2.2", "4.2.3", "4.2.4", "4.2.5"];
    const VALID_TIERS = ["FREE", "TRIAL", "PREMIUM", "PAID", "EARLY_BACKER", ""];

    if (rawTier && !VALID_TIERS.includes(rawTier)) {
      isPiratedBuild = true;
      piracyNotes.push("Illegal Tier: " + rawTier);
    }
    if (rawVersion && !AUTHORIZED_VERSIONS.includes(rawVersion.replace(/^v/i, ""))) {
      isPiratedBuild = true;
      piracyNotes.push("Unauthorized/Spoofed Version: " + rawVersion);
    }
    if (rawInstalledAt) {
      var cleanTsStr = rawInstalledAt.replace(/^FAKE_TS:/i, "").trim();
      var instNum = parseInt(cleanTsStr, 10);
      // Valid timestamps are between Jan 1 2024 (1704067200000) and Dec 31 2028 (1861833600000)
      if (isNaN(instNum) || instNum < 1704067200000 || instNum > 1861833600000) {
        isPiratedBuild = true;
        piracyNotes.push("Future/Spoofed Install Timestamp Detected");
      }
    }

    // Check for explicit Privacy Probe / Piracy Probe in status, source, term, details, or payload
    var probeContext = (status + " " + source + " " + term + " " + details + " " + rawBody + " " + (email || "")).toUpperCase();
    var isProbeDetected = isPiratedBuild || 
                          probeContext.includes("PIRACY_PROBE") || 
                          probeContext.includes("PRIVACY_PROBE") || 
                          probeContext.includes("PRIVACY PROBE") || 
                          probeContext.includes("PIRACY PROBE") ||
                          probeContext.includes("SECURITY PROBE") ||
                          probeContext.includes("TAMPERED") ||
                          probeContext.includes("CRACKED");

    // Guard against client-side false positives: If the build itself has an authorized version,
    // valid tier, and valid install timestamp, treat it as authentic even if an outdated client hook sent PIRACY_PROBE
    var isVerifiedAuthentic = !isPiratedBuild && rawVersion && AUTHORIZED_VERSIONS.includes(rawVersion.replace(/^v/i, "")) && VALID_TIERS.includes(rawTier);
    if (isVerifiedAuthentic && isProbeDetected) {
      isProbeDetected = false;
      if (status.includes("PIRACY") || status.includes("PROBE")) {
        status = rawTier === "PREMIUM" ? "UNINSTALL_PREMIUM_USER" : "UNINSTALL_PAGE_OPENED";
      }
      if (term.includes("Pirated") || term.includes("Forked")) {
        term = rawTier === "PREMIUM" ? "Premium User Uninstalled" : "Extension Uninstalled";
      }
      details = details.replace(/\[⚠️ PIRACY\/FORK PROBE DETECTED:[^\]]*\]\s*\|\s*/g, "");
      details = details.replace(/\s*\|\s*\[SUPPRESS_WINBACK:\s*Piracy Probe\]/g, "");
    }

        if (isProbeDetected) {
      status = "🚨 PIRACY_PROBE";
      details = (details ? details + " | " : "") + "⚠️ PROBE / TAMPERED BUILD DETECTED (" + (piracyNotes.length ? piracyNotes.join(", ") : "Probe Payload Detected") + ")";
    }

    // Reverse Identity Recovery for Anonymous Uninstalls
    var isAnonymous = !email || email.includes("anonymous") || !email.includes("@");
    var recoveryLog = "";

    if ((isUninstallHook || isFeedbackForm) && isAnonymous) {
      var installedAt = payload.installed_at || (e && e.parameter && e.parameter.installed_at) || extractInstalledAtFromDetails(details, rawBody);
      var recovered = recoverEmailFromInstallTimestamp(sheet, installedAt);
      if (recovered) {
        email = recovered;
        isAnonymous = false;
        recoveryLog = " [Recovered via installed_at: " + recovered + "]";
        details = (details ? details + " | " : "") + "IDENTITY_RECOVERED: " + recovered;
      }
      if (isAnonymous && rawBody) {
        details = (details ? details + " | " : "") + "RAW_PAYLOAD: " + rawBody.substring(0, 300);
      }
    }

    if (!email) {
      email = "anonymous-uninstall@kasyhq.com";
      isAnonymous = true;
    }

    // Fast Deduplication Guard: Drop duplicate event bursts within 30 seconds
    var lastRow = sheet.getLastRow();
    if (lastRow >= 2 && !isAnonymous) {
      var scanLimit = Math.min(25, lastRow - 1);
      var recentData = sheet.getRange(Math.max(2, lastRow - scanLimit + 1), 1, scanLimit, TOTAL_COLUMNS).getValues();
      for (var r = recentData.length - 1; r >= 0; r--) {
        var prevTime = parseAnyTimestamp(recentData[r][COLUMNS.TIMESTAMP - 1]);
        var prevEmail = String(recentData[r][COLUMNS.EMAIL - 1] || "").trim().toLowerCase();
        var prevStatus = String(recentData[r][COLUMNS.STATUS - 1] || "").toUpperCase();

        if (prevEmail === email && prevStatus === status && prevTime && (Math.abs(now.getTime() - prevTime.getTime()) < CONFIG.BURST_DEDUPE_WINDOW_MS)) {
          return jsonResponse({ status: "skipped_duplicate_event", email: email });
        }
      }
    }

    // Process Event via Modular Handlers
    var welcomeCol = "";
    var expiryCol = "";
    var renewalCol = "";

    if (isProbeDetected) {
      welcomeCol = "🚨 BLOCKED_PIRACY_PROBE";
      expiryCol = "BLOCKED";
      renewalCol = "BLOCKED";
    } else if (isFeedbackForm) {
      welcomeCol = STATUS.FEEDBACK_RECORDED;
    } else if (isUninstallHook) {
      if (isAnonymous) {
        // Suppress logging of anonymous uninstall page opens to the sheet
        return jsonResponse({ 
          status: "skipped_anonymous_uninstall", 
          message: "Anonymous uninstall page open not logged" 
        });
      } else {
        var isPaidSubscriber = isUserPaidInSheet(sheet, email);
        var alreadySentUninstall = hasUninstallEmailBeenSent(sheet, email);

        if (alreadySentUninstall) {
          welcomeCol = STATUS.ALREADY_SENT_UNINSTALL;
        } else {
          welcomeCol = isPaidSubscriber ? sendPaidUninstallGuideEmail(email) : sendUninstallWinbackEmail(email);
        }
      }
    } else {
      var isExisting = isUserRegisteredInSheet(sheet, email);
      var alreadySentWelcome = hasWelcomeEmailBeenSent(sheet, email);
      var alreadySentExpiry = hasExpiryEmailBeenSent(sheet, email);
      var isPaidSubscriber = isUserPaidInSheet(sheet, email);

      if (status === STATUS.NEW_USER_TRIAL) {
        if (isExisting || alreadySentWelcome) {
          status = STATUS.USER_RELOGIN;
          welcomeCol = STATUS.EXISTING_USER_RELOGIN;
          details = (details ? details + " | " : "") + "Returning user login (Original trial start date preserved)";

          // Spontaneous Expiry Check: If user trial expired (> 7 days), dispatch expiry email spontaneously
          if (!alreadySentExpiry && !isPaidSubscriber && !isAnonymous) {
            var origDate = getUserOriginalRegistrationDate(sheet, email);
            if (origDate) {
              var daysSinceStart = (now.getTime() - origDate.getTime()) / (1000 * 60 * 60 * 24);
              if (daysSinceStart >= CONFIG.TRIAL_DAYS) {
                expiryCol = sendTrialExpiryEmail(email);
                updateUserExpiryStatusInSheet(sheet, email, expiryCol);
              }
            }
          }
        } else if (!isAnonymous) {
          // Spontaneous Welcome Email: Sent immediately in real-time
          welcomeCol = sendWelcomeTrialEmail(email);
        }
      } else if (status === STATUS.USER_RELOGIN) {
        welcomeCol = STATUS.EXISTING_USER_RELOGIN;
        // Spontaneous Expiry Check on returning login
        if (!alreadySentExpiry && !isPaidSubscriber && !isAnonymous) {
          var origDate = getUserOriginalRegistrationDate(sheet, email);
          if (origDate) {
            var daysSinceStart = (now.getTime() - origDate.getTime()) / (1000 * 60 * 60 * 24);
            if (daysSinceStart >= CONFIG.TRIAL_DAYS) {
              expiryCol = sendTrialExpiryEmail(email);
              updateUserExpiryStatusInSheet(sheet, email, expiryCol);
            }
          }
        }
      } else if (status.includes("EXPIRY") || status.includes("TRIAL_EXPIRED") || status.includes("EXPIRED")) {
        // Spontaneous Trial Expiry Notification: Update existing user row without appending duplicates
        if (alreadySentExpiry || isPaidSubscriber || isAnonymous) {
          return jsonResponse({ status: "skipped_already_handled", email: email });
        }
        expiryCol = sendTrialExpiryEmail(email);
        updateUserExpiryStatusInSheet(sheet, email, expiryCol);
        return jsonResponse({ status: "success", email: email, action: expiryCol });
      } else if (status === STATUS.PAID || status === STATUS.CAPTURE_SUCCESS || status === STATUS.GIFTED) {
        welcomeCol = STATUS.PAID_USER;
        expiryCol = STATUS.PAID_USER;
      }
    }

    var newRow = [timestampStr, email, product, term, amount, status, source, device, details, welcomeCol, expiryCol, renewalCol];
    sheet.appendRow(newRow);
    SpreadsheetApp.flush();

    // Opportunistic Spontaneous Sweep: Dispatches up to 3 pending Day 7 trial expiries on every event
    try {
      sweepPendingExpiredTrials(sheet, 3);
    } catch (_) {}

    return jsonResponse({ 
      status: "success", 
      email: email, 
      action: welcomeCol,
      expiryAction: expiryCol,
      recovered: !!recoveryLog 
    });

  } catch (error) {
    return jsonResponse({ status: "error", message: error.toString() });
  } finally {
    lock.releaseLock();
  }
}

/**
 * ─────────────────────────────────────────────────────────────────────────────
 * 4. REVERSE IDENTITY RECOVERY (Strict & Safe Bounds)
 * ─────────────────────────────────────────────────────────────────────────────
 */
function recoverEmailFromInstallTimestamp(sheet, installedAtInput) {
  if (!installedAtInput) return null;
  try {
    var installMillis = parseInt(installedAtInput, 10);
    if (isNaN(installMillis)) return null;
    if (installMillis < 10000000000) {
      installMillis = installMillis * 1000;
    }
    if (installMillis < 1000000000000) return null;
    
    // Bounds check: Between Year 2020 and today + 1 day
    var nowMs = new Date().getTime();
    if (installMillis < 1577836800000 || installMillis > (nowMs + 86400000)) return null;

    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return null;

    var data = sheet.getRange(2, 1, lastRow - 1, 9).getValues();

    // Pass 1: Strict match for exact installed_at in Details column
    var targetStr = "installed_at:" + installMillis;
    var targetStrRaw = String(installMillis);

    for (var i = data.length - 1; i >= 0; i--) {
      var rowEmail = String(data[i][COLUMNS.EMAIL - 1] || "").trim().toLowerCase();
      var rowDetails = String(data[i][COLUMNS.DETAILS - 1] || "");
      if (rowEmail && rowEmail.includes("@") && !rowEmail.includes("anonymous")) {
        if (rowDetails.includes(targetStr) || (rowDetails.includes("installed_at") && rowDetails.includes(targetStrRaw))) {
          return rowEmail;
        }
      }
    }

    // Pass 2: Strict match against registration timestamp (within 2-minute window)
    var installDate = new Date(installMillis);
    var closestEmail = null;
    var smallestDiff = Infinity;

    for (var j = data.length - 1; j >= 0; j--) {
      var rowTime = parseAnyTimestamp(data[j][COLUMNS.TIMESTAMP - 1]);
      var rowEm = String(data[j][COLUMNS.EMAIL - 1] || "").trim().toLowerCase();
      var rowStatus = String(data[j][COLUMNS.STATUS - 1] || "").toUpperCase();

      if (rowTime && rowEm && rowEm.includes("@") && !rowEm.includes("anonymous")) {
        if (rowStatus.includes("TRIAL") || rowStatus === STATUS.NEW_USER_TRIAL || rowStatus.includes("PAID") || rowStatus === STATUS.GIFTED) {
          var diff = Math.abs(rowTime.getTime() - installDate.getTime());
          if (diff <= 120000 && diff < smallestDiff) { // <= 120 seconds
            smallestDiff = diff;
            closestEmail = rowEm;
          }
        }
      }
    }
    return closestEmail;
  } catch (_) {
    return null;
  }
}

function extractInstalledAtFromDetails(detailsStr, rawBodyStr) {
  try {
    var combined = (detailsStr || "") + " " + (rawBodyStr || "");
    var match = combined.match(/installed_at[=:]\s*["']?(\d{10,14})["']?/i);
    if (match && match[1]) return match[1];
    var jsonMatch = combined.match(/"installed_at"\s*:\s*"?(\d{10,14})"?/i);
    if (jsonMatch && jsonMatch[1]) return jsonMatch[1];
  } catch (_) {}
  return null;
}
 
/**
 * ─────────────────────────────────────────────────────────────────────────────
 * 4.5 SPONTANEOUS OPPORTUNISTIC EXPIRY SWEEPER (100% Triggerless)
 * Automatically called by doPost(e) on every incoming event.
 * Dispatches up to maxToSend expired trial emails spontaneously in real-time.
 * ─────────────────────────────────────────────────────────────────────────────
 */
function sweepPendingExpiredTrials(sheet, maxToSend) {
  if (!sheet) return 0;
  var limit = maxToSend || 3;
  var quota = MailApp.getRemainingDailyQuota();
  if (quota <= 0) return 0;

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return 0;

  var data = sheet.getRange(2, 1, lastRow - 1, TOTAL_COLUMNS).getValues();
  var now = new Date();
  var sent = 0;

  var paidMap = {};
  var expirySentMap = {};
  var probeMap = {};

  for (var k = 0; k < data.length; k++) {
    var em = String(data[k][COLUMNS.EMAIL - 1] || "").trim().toLowerCase();
    if (!em || !em.includes("@") || em.includes("anonymous")) continue;
    var st = String(data[k][COLUMNS.STATUS - 1] || "").toUpperCase();
    var amt = String(data[k][COLUMNS.AMOUNT - 1] || "").replace(/[^\d]/g, "");
    var exp = String(data[k][COLUMNS.EXPIRY_STATUS - 1] || "").toUpperCase();

    if (st === STATUS.PAID || st === STATUS.CAPTURE_SUCCESS || st === STATUS.GIFTED || parseInt(amt, 10) > 0) {
      paidMap[em] = true;
    }
    if (exp.includes("SENT_EXPIRY") || exp.includes("ALREADY") || exp === STATUS.PAID_USER || exp.includes("SKIPPED")) {
      expirySentMap[em] = true;
    }
    if (st.includes("PROBE") || st.includes("PIRACY")) {
      probeMap[em] = true;
    }
  }

  for (var i = 0; i < data.length && sent < limit && quota > 0; i++) {
    var email = String(data[i][COLUMNS.EMAIL - 1] || "").trim().toLowerCase();
    if (!email || !email.includes("@") || email.includes("anonymous")) continue;
    if (paidMap[email] || expirySentMap[email] || probeMap[email]) continue;

    var status = String(data[i][COLUMNS.STATUS - 1] || "").toUpperCase();
    var isTrialRow = status === STATUS.NEW_USER_TRIAL || status.includes("TRIAL") || status === STATUS.USER_RELOGIN;
    if (!isTrialRow) continue;

    var rawTs = data[i][COLUMNS.TIMESTAMP - 1];
    var parsed = parseAnyTimestamp(rawTs);
    if (!parsed) continue;

    var expiryDate = new Date(parsed.getTime() + (CONFIG.TRIAL_DAYS * 24 * 60 * 60 * 1000));
    if (now >= expiryDate) {
      var hoursSinceExpiry = (now.getTime() - expiryDate.getTime()) / (1000 * 60 * 60);
      var maxHours = CONFIG.EXPIRY_WINDOW_HOURS || 120; // 5 days
      var resStatus = "";

      if (hoursSinceExpiry <= maxHours) {
        resStatus = sendTrialExpiryEmail(email);
        if (String(resStatus).startsWith("SENT_EXPIRY")) {
          sent++;
          quota--;
        }
      } else {
        resStatus = STATUS.SKIPPED_HISTORICAL_EXPIRY;
      }

      sheet.getRange(i + 2, COLUMNS.EXPIRY_STATUS).setValue(resStatus);
      expirySentMap[email] = true;
    }
  }

  return sent;
}

/**
 * ─────────────────────────────────────────────────────────────────────────────
 * 5. MANUAL LIFECYCLE SWEEP & CATCHUP ENGINE (No Background Triggers Required)
 * ─────────────────────────────────────────────────────────────────────────────
 */
function processTrialEmails() {
  processAllLifecycleEmails();
}

function processAllLifecycleEmails() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getRawDataSheet(ss);
  if (!sheet || sheet.getLastRow() < 2) return;

  var remainingQuota = MailApp.getRemainingDailyQuota();
  if (remainingQuota <= 0) {
    SpreadsheetApp.getActiveSpreadsheet().toast("Gmail daily quota exhausted (0 remaining). Aborting run.", "Quota Exhausted", 6);
    return;
  }

  var startTime = new Date().getTime();
  var lastRow = sheet.getLastRow();
  var totalRows = lastRow - 1;
  var data = sheet.getRange(2, 1, totalRows, TOTAL_COLUMNS).getValues();
  var now = new Date();

  var statusUpdates = [];
  var sentCount = 0;

  // 1. Build in-memory lookup maps to guarantee 1-email limit
  var trulySentWelcomeEmails = {};
  var trulySentUninstallEmails = {};
  var paidEmailsMap = {};
  var blockedProbeEmails = {};

  for (var k = 0; k < totalRows; k++) {
    var kEmail = String(data[k][COLUMNS.EMAIL - 1] || "").trim().toLowerCase();
    var kStatus = String(data[k][COLUMNS.STATUS - 1] || "").toUpperCase();
    var kDetails = String(data[k][COLUMNS.DETAILS - 1] || "").toUpperCase();
    var kAmount = String(data[k][COLUMNS.AMOUNT - 1] || "").replace(/[^\d]/g, "");
    var kWel = String(data[k][COLUMNS.WELCOME_STATUS - 1] || "").trim().toUpperCase();

    if (kEmail && !kEmail.includes("anonymous") && kEmail.includes("@")) {
      var isPaidEvent = kStatus === STATUS.PAID || kStatus === STATUS.CAPTURE_SUCCESS || kStatus === STATUS.GIFTED || parseInt(kAmount, 10) > 0;
      var isNotTrialUninstall = !kStatus.includes("UNINSTALL") && !kStatus.includes("TRIAL") && kStatus !== "ORDER_CREATED";
      if (isPaidEvent && isNotTrialUninstall) {
        paidEmailsMap[kEmail] = true;
      }

      var isRowProbe = !isPaidEvent && (
        kStatus.includes("PROBE") || kStatus.includes("PIRACY") || kStatus.includes("PRIVACY") ||
        kWel.includes("PROBE") || (kWel.includes("BLOCKED") && !kWel.includes("PAID")) || 
        kDetails.includes("PROBE") || kDetails.includes("TAMPERED") || kDetails.includes("CRACKED")
      );
      if (isRowProbe) {
        blockedProbeEmails[kEmail] = true;
      }
      if (isWelcomeSentStatus(kWel)) trulySentWelcomeEmails[kEmail] = true;
      if (kWel.includes("WINBACK") || kWel.includes("GUIDE") || kWel.includes("ALREADY_SENT_UNINSTALL")) {
        trulySentUninstallEmails[kEmail] = true;
      }
    }
  }

  // 2. Process rows
  for (var i = 0; i < totalRows; i++) {
    // Timeout safeguard (exit gracefully if approaching 4.5 minutes)
    if (new Date().getTime() - startTime > CONFIG.MAX_SCRIPT_EXEC_MS) {
      break;
    }

    var rawTimestamp = data[i][COLUMNS.TIMESTAMP - 1];
    var email = String(data[i][COLUMNS.EMAIL - 1] || "").trim().toLowerCase();
    var term = String(data[i][COLUMNS.TERM - 1] || "").toUpperCase();
    var status = String(data[i][COLUMNS.STATUS - 1] || "").toUpperCase();
    var source = String(data[i][COLUMNS.SOURCE - 1] || "").toUpperCase();
    var details = String(data[i][COLUMNS.DETAILS - 1] || "").toUpperCase();
    
    var welcomeSent = String(data[i][COLUMNS.WELCOME_STATUS - 1] || "").trim();
    var expirySent = String(data[i][COLUMNS.EXPIRY_STATUS - 1] || "").trim();
    var renewalSent = String(data[i][COLUMNS.RENEWAL_STATUS - 1] || "").trim();

    if (remainingQuota <= 0 || !email || !email.includes("@") || email.includes("anonymous")) {
      statusUpdates.push([welcomeSent, expirySent, renewalSent]);
      continue;
    }

    // FLOW 0: PROBE & PIRACY SAFEGUARD (NEVER SENDS ANY AUTOMATIC EMAILS)
    var isProbeRow = status.includes("PROBE") || status.includes("PIRACY") || status.includes("PRIVACY") ||
                     details.includes("PROBE") || details.includes("TAMPERED") || details.includes("CRACKED") ||
                     welcomeSent.includes("PROBE") || welcomeSent.includes("BLOCKED") ||
                     blockedProbeEmails[email] === true;

    if (isProbeRow) {
      if (!welcomeSent || !welcomeSent.includes("BLOCKED")) welcomeSent = "🚨 BLOCKED_PIRACY_PROBE";
      if (!expirySent || expirySent === "") expirySent = "BLOCKED";
      if (!renewalSent || renewalSent === "") renewalSent = "BLOCKED";
      statusUpdates.push([welcomeSent, expirySent, renewalSent]);
      continue;
    }

    var isUserPaid = paidEmailsMap[email] === true;

    // FLOW A: FEEDBACK ROWS (SURVEY ONLY — NEVER SENDS EMAIL)
    var isFeedbackRow = status === STATUS.FEEDBACK_SUBMITTED || source.includes("FORM") || term.includes("FEEDBACK") || details.includes("REASON:");
    if (isFeedbackRow) {
      if (!welcomeSent) welcomeSent = STATUS.FEEDBACK_RECORDED;
      statusUpdates.push([welcomeSent, expirySent, renewalSent]);
      continue;
    }

    // FLOW B: UNINSTALL ROWS
    var isUninstallRow = status.includes("UNINSTALL") || details.includes("EXTENSION REMOVED") || details.includes("UNINSTALLED") || term.includes("UNINSTALL");
    if (isUninstallRow) {
      if (!isWelcomeSentStatus(welcomeSent)) {
        if (trulySentUninstallEmails[email]) {
          welcomeSent = STATUS.ALREADY_SENT_UNINSTALL;
        } else {
          welcomeSent = isUserPaid ? sendPaidUninstallGuideEmail(email) : sendUninstallWinbackEmail(email);
          if (welcomeSent.includes("SENT")) { 
            sentCount++; 
            remainingQuota--; 
            trulySentUninstallEmails[email] = true; 
          }
        }
      }
      statusUpdates.push([welcomeSent, expirySent, renewalSent]);
      continue;
    }

    var parsedDate = parseAnyTimestamp(rawTimestamp);

    // FLOW C: TRIAL LIFECYCLE (Day 0 Welcome Catchup & Day 7 Expiry)
    if (!isUserPaid && (status.includes("TRIAL") || term.includes("TRIAL") || status === STATUS.NEW_USER_TRIAL || status === STATUS.USER_RELOGIN)) {
      var ageHours = parsedDate ? (now.getTime() - parsedDate.getTime()) / (1000 * 60 * 60) : 0;

      // Welcome Email Catchup
      if (!isWelcomeSentStatus(welcomeSent)) {
        if (trulySentWelcomeEmails[email]) {
          welcomeSent = STATUS.ALREADY_SENT_EARLIER;
        } else if (parsedDate && ageHours > 72) {
          welcomeSent = STATUS.SKIPPED_OLD_TRIAL;
          trulySentWelcomeEmails[email] = true;
        } else if (remainingQuota > 0) {
          welcomeSent = sendWelcomeTrialEmail(email);
          if (welcomeSent.startsWith("SENT_WELCOME")) {
            trulySentWelcomeEmails[email] = true;
            sentCount++;
            remainingQuota--;
          }
        }
      }

      // Day 7 Trial Expiry Email
      var hasExpiryBeenSent = expirySent.startsWith("SENT_EXPIRY") || expirySent === STATUS.PAID_USER || expirySent === STATUS.SKIPPED_HISTORICAL;
      if (!hasExpiryBeenSent && parsedDate && remainingQuota > 0) {
        var trialExpiryDate = new Date(parsedDate.getTime() + (CONFIG.TRIAL_DAYS * 24 * 60 * 60 * 1000));
        var hoursSinceExpiry = (now.getTime() - trialExpiryDate.getTime()) / (1000 * 60 * 60);

        if (now >= trialExpiryDate) {
          var maxExpiryHours = CONFIG.EXPIRY_WINDOW_HOURS || 120;
          if (hoursSinceExpiry <= maxExpiryHours) {
            expirySent = sendTrialExpiryEmail(email);
            if (expirySent.startsWith("SENT_EXPIRY")) { sentCount++; remainingQuota--; }
          } else {
            expirySent = STATUS.SKIPPED_HISTORICAL_EXPIRY;
          }
        }
      }
    }

    // FLOW D: PAID / VIP RENEWALS (3-Day Warning & Day-Of Expiry)
    if (isUserPaid && (status === STATUS.PAID || status === STATUS.CAPTURE_SUCCESS || status === STATUS.GIFTED || 
        ((term.includes("MONTH") || term.includes("YEAR") || term.includes("QUARTER")) && !term.includes("TRIAL")))) {
      if (!welcomeSent) welcomeSent = STATUS.PAID_USER;
      if (!expirySent) expirySent = STATUS.PAID_USER;

      if (parsedDate && remainingQuota > 0) {
        var durationDays = 30;
        if (term.includes("YEAR") || term.includes("ANNUAL") || term.includes("365")) {
          durationDays = 365;
        } else if (term.includes("QUARTER") || term.includes("3 MONTH") || term.includes("90") || term.includes("399")) {
          durationDays = 90;
        }

        var paidExpiryDate = new Date(parsedDate.getTime() + (durationDays * 24 * 60 * 60 * 1000));
        var daysUntilExpiry = (paidExpiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24);
        var hasRenewalBeenSent = renewalSent.startsWith("SENT_RENEWAL") || renewalSent.startsWith("SENT_EXPIRED") || renewalSent.includes("EXPIRED");

        if (!hasRenewalBeenSent && daysUntilExpiry <= CONFIG.RENEWAL_NOTICE_DAYS && daysUntilExpiry > 0) {
          renewalSent = sendPaidRenewalWarningEmail(email, daysUntilExpiry, paidExpiryDate);
          if (renewalSent.startsWith("SENT_RENEWAL")) { sentCount++; remainingQuota--; }
        } else if (now >= paidExpiryDate && !hasRenewalBeenSent) {
          var hoursSincePaidExpiry = (now.getTime() - paidExpiryDate.getTime()) / (1000 * 60 * 60);
          if (hoursSincePaidExpiry <= 72) {
            renewalSent = sendPaidExpiredEmail(email, paidExpiryDate);
            if (renewalSent.startsWith("SENT_EXPIRED")) { sentCount++; remainingQuota--; }
          } else {
            renewalSent = STATUS.SKIPPED_HISTORICAL_RENEWAL;
          }
        }
      }
    }

    statusUpdates.push([welcomeSent, expirySent, renewalSent]);
  }

  if (statusUpdates.length > 0) {
    sheet.getRange(2, COLUMNS.WELCOME_STATUS, statusUpdates.length, 3).setValues(statusUpdates);
    SpreadsheetApp.flush();
  }

  SpreadsheetApp.getActiveSpreadsheet().toast("Lifecycle complete. Sent " + sentCount + " emails. Quota remaining: " + remainingQuota, "Complete", 5);
}

function isWelcomeSentStatus(statusStr) {
  if (!statusStr) return false;
  var s = String(statusStr).trim().toUpperCase();
  return s.startsWith("SENT") || 
         s.startsWith("WELCOME") || 
         s.startsWith("GUIDE") || 
         s.startsWith("WINBACK") || 
         s.startsWith("ALREADY") || 
         s.startsWith("SKIPPED") ||
         s.startsWith("FEEDBACK") ||
         s.includes("BLOCKED") ||
         s.includes("PROBE") ||
         s === STATUS.PAID_USER || 
         s === STATUS.PREMIUM_USER;
}

function dispatchEmail(email, subject, html) {
  if (!email || !email.includes("@") || email.includes("anonymous")) return "SKIPPED_ANONYMOUS";
  var cleanEm = String(email).trim().toLowerCase();
  if (cleanEm.includes("probe") || cleanEm.includes("piracy") || cleanEm.includes("privacy_probe") || cleanEm.includes("piracy_probe")) {
    return "SKIPPED_PROBE_BLOCKED";
  }

  // Production Account Guard: Strictly block email dispatch if executing under legacy personal account
  var executingUser = Session.getEffectiveUser().getEmail().toLowerCase();
  if (executingUser && executingUser !== CONFIG.FROM_EMAIL.toLowerCase()) {
    Logger.log("Blocked email dispatch: Script executing under non-production account (" + executingUser + "). Only " + CONFIG.FROM_EMAIL + " is authorized to send emails.");
    return "BLOCKED_NON_PRODUCTION_ACCOUNT: " + executingUser;
  }

  try {
    GmailApp.sendEmail(cleanEm, subject, "", {
      name: CONFIG.SENDER_NAME,
      from: CONFIG.FROM_EMAIL,
      replyTo: CONFIG.REPLY_TO,
      htmlBody: html
    });
    return true;
  } catch (e) {
    return e.toString();
  }
}

/**
 * ─────────────────────────────────────────────────────────────────────────────
 * 8. MIS BEAUTIFIER & AUTO-FORMATTER
 * ─────────────────────────────────────────────────────────────────────────────
 */
function formatAndBeautifySheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getRawDataSheet(ss);
  if (!sheet) return;

  var lastRow = sheet.getLastRow();
  if (lastRow < 1) return;

  createStandardHeaders(sheet);

  if (lastRow >= 2) {
    var totalRows = lastRow - 1;
    var rangeAll = sheet.getRange(2, 1, totalRows, TOTAL_COLUMNS);
    
    rangeAll.setFontFamily("Segoe UI")
            .setFontSize(9.5)
            .setVerticalAlignment("middle");

    // Standardize Timestamps in Column A across all rows
    var colATimestamps = sheet.getRange(2, 1, totalRows, 1).getValues();
    for (var r = 0; r < totalRows; r++) {
      var d = parseAnyTimestamp(colATimestamps[r][0]);
      if (d) {
        colATimestamps[r][0] = formatToStandardIst(d);
      }
    }
    sheet.getRange(2, 1, totalRows, 1).setValues(colATimestamps);

    // Standardize Amounts in Column E to clean numeric values with Currency Format
    var colEAmounts = sheet.getRange(2, COLUMNS.AMOUNT, totalRows, 1).getValues();
    for (var a = 0; a < totalRows; a++) {
      var val = colEAmounts[a][0];
      if (val !== "" && val !== null && val !== undefined) {
        var numStr = String(val).replace(/[^\d.]/g, "");
        if (numStr) {
          colEAmounts[a][0] = parseFloat(numStr);
        }
      }
    }
    sheet.getRange(2, COLUMNS.AMOUNT, totalRows, 1).setValues(colEAmounts).setNumberFormat('"₹"#,##0');

    // Alignments
    sheet.getRange(2, COLUMNS.TIMESTAMP, totalRows, 1).setHorizontalAlignment("center");
    sheet.getRange(2, COLUMNS.EMAIL, totalRows, 1).setHorizontalAlignment("left");
    sheet.getRange(2, COLUMNS.PRODUCT, totalRows, 2).setHorizontalAlignment("center");
    sheet.getRange(2, COLUMNS.AMOUNT, totalRows, 1).setHorizontalAlignment("right").setFontWeight("bold");
    sheet.getRange(2, COLUMNS.STATUS, totalRows, 3).setHorizontalAlignment("center");
    sheet.getRange(2, COLUMNS.DETAILS, totalRows, 1).setHorizontalAlignment("left");
    sheet.getRange(2, COLUMNS.WELCOME_STATUS, totalRows, 3).setHorizontalAlignment("center").setFontSize(8.5);

    setupConditionalFormatting(sheet);

    sheet.setColumnWidth(COLUMNS.TIMESTAMP, 155);
    sheet.setColumnWidth(COLUMNS.EMAIL, 230);
    sheet.setColumnWidth(COLUMNS.PRODUCT, 110);
    sheet.setColumnWidth(COLUMNS.TERM, 120);
    sheet.setColumnWidth(COLUMNS.AMOUNT, 95);
    sheet.setColumnWidth(COLUMNS.STATUS, 160);
    sheet.setColumnWidth(COLUMNS.SOURCE, 130);
    sheet.setColumnWidth(COLUMNS.DEVICE, 120);
    sheet.setColumnWidth(COLUMNS.DETAILS, 240);
    sheet.setColumnWidth(COLUMNS.WELCOME_STATUS, 180);
    sheet.setColumnWidth(COLUMNS.EXPIRY_STATUS, 160);
    sheet.setColumnWidth(COLUMNS.RENEWAL_STATUS, 180);
  }

  SpreadsheetApp.getActiveSpreadsheet().toast("MIS Sheet cleaned, standardized, and formatted successfully!", "✨ Beautify Complete", 5);
}

/**
 * ─────────────────────────────────────────────────────────────────────────────
 * 9. FIREBASE SYNC (One-Time Catchup)
 * ─────────────────────────────────────────────────────────────────────────────
 */
function syncAndPopulateTrials() {
  var apiUrl = "https://kasyhq.com/api/admin/subscribers?secret=" + encodeURIComponent(CONFIG.ADMIN_SECRET_KEY);
  try {
    var response = UrlFetchApp.fetch(apiUrl, { muteHttpExceptions: true });
    var json = JSON.parse(response.getContentText());

    if (!json.success || !json.subscribers) {
      SpreadsheetApp.getUi().alert("Sync Notice: " + (json.error || "Please verify ADMIN_SECRET_KEY in script."));
      return;
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = getRawDataSheet(ss);

    var existingEmails = {};
    if (sheet.getLastRow() > 1) {
      var oldRows = sheet.getRange(2, COLUMNS.EMAIL, sheet.getLastRow() - 1, 1).getValues();
      oldRows.forEach(function(r) {
        var em = String(r[0]).trim().toLowerCase();
        if (em) existingEmails[em] = true;
      });
    }

    var newRowsToAppend = [];
    json.subscribers.forEach(function(s) {
      var email = String(s.email || "").trim().toLowerCase();
      if (email && !existingEmails[email]) {
        var dateObj = s.firstSeenAt ? new Date(s.firstSeenAt) : new Date();
        var dateStr = formatToStandardIst(dateObj);
        
        var isVip = s.category === "GIFTED" || s.category === "PAID" || String(s.plan || "").toUpperCase() === "PREMIUM";
        var statusLabel = s.category === "GIFTED" ? STATUS.GIFTED : (s.category === "PAID" ? STATUS.PAID : STATUS.NEW_USER_TRIAL);
        var termLabel = s.billingCycle || (isVip ? "PREMIUM LICENSE" : "Trial (7 Days)");
        
        newRowsToAppend.push([
          dateStr,
          email,
          s.product || "ReturnPilot-GST Return Downloader",
          termLabel,
          s.amount || "",
          statusLabel,
          "Firebase Sync",
          "Web",
          "Imported from Firestore: " + (s.plan || "FREE"),
          isVip ? STATUS.PREMIUM_USER : STATUS.SKIPPED_HISTORICAL,
          isVip ? STATUS.PREMIUM_USER : "",
          ""
        ]);
        existingEmails[email] = true;
      }
    });

    if (newRowsToAppend.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, newRowsToAppend.length, TOTAL_COLUMNS).setValues(newRowsToAppend);
      SpreadsheetApp.flush();
      formatAndBeautifySheet();
    }

    SpreadsheetApp.getActiveSpreadsheet().toast("Synced " + newRowsToAppend.length + " users from Firebase!", "Sync Complete", 5);
  } catch (err) {
    Logger.log("Sync error: " + err.toString());
  }
}

 * 10. TIMEZONE-SAFE DATE PARSERS & FORMATTERS
 * ─────────────────────────────────────────────────────────────────────────────

function formatToStandardIst(d) {
  if (!d || !(d instanceof Date) || isNaN(d.getTime())) return "";
  return Utilities.formatDate(d, CONFIG.TIMEZONE, "dd-MMM-yyyy HH:mm:ss");
}

function formatToShortDate(d) {
  if (!d || !(d instanceof Date) || isNaN(d.getTime())) return "";
  return Utilities.formatDate(d, CONFIG.TIMEZONE, "dd/MM/yyyy");
}

function parseAnyTimestamp(input) {
  if (!input) return null;
  if (input instanceof Date) return isNaN(input.getTime()) ? null : input;

  var str = String(input).trim();
  if (!str) return null;

  // Epoch ms or seconds
  if (/^\d{10,14}$/.test(str)) {
    var num = parseInt(str, 10);
    if (str.length === 10) num *= 1000;
    var dNum = new Date(num);
    return isNaN(dNum.getTime()) ? null : dNum;
  }

  // ISO 8601 strings
  if (str.includes("T") && (str.endsWith("Z") || str.includes("+") || str.includes("-"))) {
    var dIso = new Date(str);
    if (!isNaN(dIso.getTime())) return dIso;
  }

  var monthMap = { 
    jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5, 
    jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11 
  };

  try {
    var cleanStr = str.replace(/,/g, " ").replace(/\s+/g, " ").trim();
    var parts = cleanStr.split(" ");
    if (parts.length >= 1) {
      var dateParts = parts[0].split(/[\/\-]/);
      var timeParts = parts.length > 1 ? parts[1].split(":") : ["0", "0", "0"];
      var ampm = "";

      for (var p = 1; p < parts.length; p++) {
        var lower = parts[p].toLowerCase();
        if (lower === "am" || lower === "pm") {
          ampm = lower;
          break;
        }
      }

      var day = parseInt(dateParts[0], 10);
      var monthRaw = String(dateParts[1] || "").toLowerCase();
      var month = monthMap[monthRaw] !== undefined ? monthMap[monthRaw] : (parseInt(monthRaw, 10) - 1);
      var year = parseInt(dateParts[2], 10);
      if (year < 100) year += 2000;

      var hour = timeParts.length > 0 ? parseInt(timeParts[0], 10) : 0;
      var min = timeParts.length > 1 ? parseInt(timeParts[1], 10) : 0;
      var sec = timeParts.length > 2 ? parseInt(timeParts[2], 10) : 0;

      if (ampm === "pm" && hour < 12) {
        hour += 12;
      } else if (ampm === "am" && hour === 12) {
        hour = 0;
      }

      // Convert IST date string (UTC+5:30) to exact epoch timestamp
      var utcMillis = Date.UTC(year, month, day, hour, min, sec) - (5.5 * 60 * 60 * 1000);
      var parsed = new Date(utcMillis);
      return isNaN(parsed.getTime()) ? null : parsed;
    }
  } catch (_) {}
  return null;
}
